@app.route('/api/products')
def products():
    try:
        # Получаем параметры фильтрации
        type_id = request.args.get('type_id', type=int)
        flag_ids = request.args.getlist('flag_ids', type=int)

        conn = get_db_connection()
        cur = conn.cursor()

        # Подзапрос для исключения товаров с флагом "Скрытый" (регистронезависимо)
        hidden_flag_condition = """
            NOT EXISTS (
                SELECT 1 FROM flags_for_products fp
                JOIN flags f ON fp.flags_id = f.id
                WHERE fp.products_id = p.id AND lower(f.name) = 'скрытый'
            )
        """

        if flag_ids:
            # Формируем запрос: товары, у которых есть ВСЕ указанные флаги, и нет скрытого флага
            query = f"""
                SELECT p.code, p.name
                FROM products p
                JOIN flags_for_products fp ON p.id = fp.products_id
                WHERE 1=1
                AND {hidden_flag_condition}
            """
            params = []
            if type_id is not None:
                query += " AND p.products_type_id = %s"
                params.append(type_id)
            query += " AND fp.flags_id = ANY(%s)"
            params.append(flag_ids)
            query += """
                GROUP BY p.id, p.code, p.name
                HAVING COUNT(DISTINCT fp.flags_id) = %s
                ORDER BY p.id
            """
            params.append(len(flag_ids))
            cur.execute(query, params)
        else:
            # Без фильтра по флагам, но исключаем скрытые товары
            query = f"""
                SELECT code, name
                FROM products p
                WHERE 1=1
                AND {hidden_flag_condition}
            """
            params = []
            if type_id is not None:
                query += " AND products_type_id = %s"
                params.append(type_id)
            query += " ORDER BY id"
            cur.execute(query, params)

        rows = cur.fetchall()
        cur.close()
        conn.close()
        return jsonify([{'code': r[0], 'name': r[1]} for r in rows])
    except Exception as e:
        app.logger.error(f"API error: {e}")
        return jsonify({'error': 'Internal server error'}), 500

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import os
import sys

# Расширения файлов, которые считаем текстовыми
TEXT_EXTENSIONS = {
    '.txt', '.html', '.htm', '.css', '.js', '.py', '.json',
    '.xml', '.csv', '.md', '.rst', '.ini', '.cfg', '.conf',
    '.yaml', '.yml', '.toml', '.sh', '.bat', '.ps1', '.rb',
    '.pl', '.php', '.java', '.c', '.cpp', '.h', '.hpp', '.go',
    '.rs', '.swift', '.kt', '.scala', '.sql', '.r', '.jl'
}

# Имя выходного файла
OUTPUT_FILE = 'code.txt'

def collect_text_files(root_dir, output_file):
    """
    Рекурсивно обходит root_dir, находит текстовые файлы с заданными расширениями
    и записывает их содержимое в output_file с указанием относительного пути.
    """
    script_dir = os.path.dirname(os.path.abspath(__file__))
    # Используем root_dir как отправную точку для обхода
    # Если root_dir не указан, используем директорию скрипта
    if root_dir is None:
        root_dir = script_dir
    else:
        root_dir = os.path.abspath(root_dir)

    output_path = os.path.join(root_dir, output_file)

    # Счётчики для статистики
    processed = 0
    skipped = 0
    errors = 0

    with open(output_path, 'w', encoding='utf-8') as out_f:
        for dirpath, dirnames, filenames in os.walk(root_dir):
            # Исключаем выходной файл, если он попадёт в обход
            if output_file in filenames:
                filenames.remove(output_file)

            for filename in filenames:
                file_path = os.path.join(dirpath, filename)
                # Проверяем расширение
                ext = os.path.splitext(filename)[1].lower()
                if ext not in TEXT_EXTENSIONS:
                    skipped += 1
                    continue

                # Относительный путь от корня обхода (root_dir)
                rel_path = os.path.relpath(file_path, start=root_dir)
                # Если root_dir совпадает с директорией скрипта, то rel_path – это
                # путь от корня проекта (места запуска) до файла

                try:
                    # Пытаемся прочитать файл с кодировкой UTF-8
                    with open(file_path, 'r', encoding='utf-8') as in_f:
                        content = in_f.read()
                except UnicodeDecodeError:
                    # Если не UTF-8, пробуем другие распространённые кодировки
                    try:
                        with open(file_path, 'r', encoding='cp1251') as in_f:
                            content = in_f.read()
                    except Exception as e:
                        errors += 1
                        print(f"Ошибка чтения {file_path}: {e}", file=sys.stderr)
                        content = f"[Ошибка: невозможно прочитать файл в текстовом режиме]"
                except Exception as e:
                    errors += 1
                    print(f"Ошибка при открытии {file_path}: {e}", file=sys.stderr)
                    continue

                # Записываем в выходной файл
                out_f.write(f"===== Файл: {rel_path} =====\n")
                out_f.write(content)
                # Добавляем пустую строку между файлами для читаемости, если содержимое не заканчивается переводом строки
                if not content.endswith('\n'):
                    out_f.write('\n')
                out_f.write('\n')  # дополнительный разделитель

                processed += 1

    print(f"Готово. Обработано файлов: {processed}, пропущено (неподходящее расширение): {skipped}, ошибок: {errors}")
    print(f"Результат записан в: {output_path}")

if __name__ == '__main__':
    # Если передан аргумент командной строки – используем его как корневую директорию
    # Иначе обходим директорию, в которой находится скрипт
    root = sys.argv[1] if len(sys.argv) > 1 else None
    collect_text_files(root, OUTPUT_FILE)

        return await func(update, context)

                if search:
                    query += f" AND (name ILIKE ${idx} OR code ILIKE ${idx})"
                    params.append(f"%{search}%")
                    idx += 1
                if type_id is not None:
                    query += f" AND products_type_id = ${idx}"
                    params.append(type_id)
                    idx += 1
                query += f" ORDER BY id OFFSET ${idx} LIMIT ${idx+1}"
                params.append(offset)
                params.append(limit)
                total = await conn.fetchval(
                    "SELECT COUNT(*) FROM products WHERE 1=1" +
                    (f" AND (name ILIKE '%{search}%' OR code ILIKE '%{search}%')" if search else "") +
                    (f" AND products_type_id = {type_id}" if type_id else "")
                )
            rows = await conn.fetch(query, *params)
            return [dict(r) for r in rows], total

    async def get_product_by_id(self, prod_id: int):
        async with self.pool.acquire() as conn:
            row = await conn.fetchrow("SELECT id, code, name, products_type_id FROM products WHERE id = $1", prod_id)
            return dict(row) if row else None

    async def get_product_by_code(self, code: str):
        async with self.pool.acquire() as conn:
            row = await conn.fetchrow("SELECT id, code, name, products_type_id FROM products WHERE code = $1", code)
            return dict(row) if row else None

    async def create_product(self, code: str, name: str, type_id: int) -> int:
        async with self.pool.acquire() as conn:
            return await conn.fetchval(
                "INSERT INTO products (code, name, products_type_id) VALUES ($1, $2, $3) RETURNING id",
                code, name, type_id
            )

    async def update_product(self, prod_id: int, code: str, name: str, type_id: int):
        async with self.pool.acquire() as conn:
            await conn.execute(
                "UPDATE products SET code = $1, name = $2, products_type_id = $3 WHERE id = $4",
                code, name, type_id, prod_id
            )

    async def delete_product_cascade(self, prod_id: int):
        async with self.pool.acquire() as conn:
            async with conn.transaction():
                await conn.execute("DELETE FROM flags_for_products WHERE products_id = $1", prod_id)
                await conn.execute("DELETE FROM materials_for_products WHERE products_id = $1", prod_id)
                await conn.execute("DELETE FROM product_main_features WHERE products_id = $1", prod_id)
                await conn.execute("DELETE FROM product_extra_features WHERE products_id = $1", prod_id)
                await conn.execute("DELETE FROM products WHERE id = $1", prod_id)

    async def get_product_flags(self, prod_id: int):
        async with self.pool.acquire() as conn:
            rows = await conn.fetch(
                "SELECT f.id, f.name FROM flags f JOIN flags_for_products fp ON f.id = fp.flags_id "
                "WHERE fp.products_id = $1 ORDER BY f.id", prod_id
            )
            return [dict(r) for r in rows]

    async def set_product_flags(self, prod_id: int, flag_ids: List[int]):
        async with self.pool.acquire() as conn:
            async with conn.transaction():
                await conn.execute("DELETE FROM flags_for_products WHERE products_id = $1", prod_id)
                if flag_ids:
                    await conn.executemany(
                        "INSERT INTO flags_for_products (products_id, flags_id) VALUES ($1, $2)",
                        [(prod_id, fid) for fid in flag_ids]
                    )

    async def get_costs_for_product(self, prod_id: int):
        async with self.pool.acquire() as conn:
            rows = await conn.fetch(
                "SELECT mfp.id, mfp.materials_type_id, mt.name as material_name, mfp.cost "
                "FROM materials_for_products mfp JOIN materials_type mt ON mfp.materials_type_id = mt.id "
                "WHERE mfp.products_id = $1 ORDER BY mt.id", prod_id
            )
            return [dict(r) for r in rows]

    async def set_cost(self, prod_id: int, material_type_id: int, cost: int):
        async with self.pool.acquire() as conn:
            await conn.execute(
                "INSERT INTO materials_for_products (products_id, materials_type_id, cost) VALUES ($1, $2, $3) "
                "ON CONFLICT (products_id, materials_type_id) DO UPDATE SET cost = $3",
                prod_id, material_type_id, cost
            )

    async def delete_cost(self, cost_id: int):
        async with self.pool.acquire() as conn:
            await conn.execute("DELETE FROM materials_for_products WHERE id = $1", cost_id)

    async def is_code_unique(self, code: str, exclude_product_id: Optional[int] = None) -> bool:
        async with self.pool.acquire() as conn:
            if exclude_product_id:
                row = await conn.fetchrow("SELECT id FROM products WHERE code = $1 AND id != $2", code, exclude_product_id)
            else:
                row = await conn.fetchrow("SELECT id FROM products WHERE code = $1", code)
            return row is None

async def download_photo_with_retry(photo, max_retries=3, delay=2.0):
    for attempt in range(max_retries):
        try:
            return await photo.get_file()
        except Exception as e:
            logger.warning(f"Попытка {attempt+1}/{max_retries} скачать фото не удалась: {e}")
            if attempt == max_retries - 1:
                raise
            await asyncio.sleep(delay)
    return None

def get_product_media_dir(product_code: str) -> Path:
    return MEDIA_BASE / product_code

def ensure_media_dir(product_code: str) -> Path:
    path = get_product_media_dir(product_code)
    path.mkdir(parents=True, exist_ok=True)
    return path

async def save_photo_from_telegram(file, product_code: str, photo_type: str, index: int = None) -> bool:
    try:
        photo_bytes = await file.download_as_bytearray()
        img = Image.open(io.BytesIO(photo_bytes))
        if img.mode in ('RGBA', 'LA', 'P'):
            rgb_img = Image.new('RGB', img.size, (255, 255, 255))
            rgb_img.paste(img, mask=img.split()[-1] if img.mode == 'RGBA' else None)
            img = rgb_img
        ensure_media_dir(product_code)
        if photo_type == 'preview':
            filename = 'preview.webp'
            quality = 50
        elif photo_type == 'size':
            filename = 'size.webp'
            quality = 85
        else:
            if index is None:
                existing = list(get_product_media_dir(product_code).glob('[0-9]*.webp'))
                numbers = [int(p.stem) for p in existing if p.stem.isdigit()]
                index = max(numbers) + 1 if numbers else 1
            filename = f'{index}.webp'
            quality = 85
        filepath = get_product_media_dir(product_code) / filename
        img.save(filepath, 'WEBP', quality=quality)
        return True
    except Exception as e:
        logger.error(f"Ошибка сохранения фото: {e}")
        return False

async def delete_photo(product_code: str, photo_type: str, index: int = None) -> bool:
    if photo_type == 'preview':
        filename = 'preview.webp'
    elif photo_type == 'size':
        filename = 'size.webp'
    else:
        if index is None:
            return False
        filename = f'{index}.webp'
    filepath = get_product_media_dir(product_code) / filename
    if not filepath.exists():
        return False
    filepath.unlink()
    if photo_type == 'main':
        base_dir = get_product_media_dir(product_code)
        main_files = sorted([p for p in base_dir.glob('[0-9]*.webp') if p.stem.isdigit()], key=lambda x: int(x.stem))
        new_index = 1
        for p in main_files:
            current_num = int(p.stem)
            if current_num != new_index:
                new_path = base_dir / f'{new_index}.webp'
                p.rename(new_path)
            new_index += 1
    return True

def get_product_photos(product_code: str):
    base = get_product_media_dir(product_code)
    if not base.exists():
        return {'preview': None, 'size': None, 'main': []}
    return {
        'preview': str(base / 'preview.webp') if (base / 'preview.webp').exists() else None,
        'size': str(base / 'size.webp') if (base / 'size.webp').exists() else None,
        'main': sorted([str(p) for p in base.glob('[0-9]*.webp') if p.stem.isdigit()], key=lambda x: int(Path(x).stem))
    }

async def move_media_folder(old_code: str, new_code: str) -> bool:
    old_path = get_product_media_dir(old_code)
    new_path = get_product_media_dir(new_code)
    if old_path.exists():
        if new_path.exists():
            for item in old_path.iterdir():
                shutil.move(str(item), str(new_path / item.name))
            old_path.rmdir()
        else:
            shutil.move(str(old_path), str(new_path))
        return True
    return False

async def delete_media_folder(product_code: str):
    path = get_product_media_dir(product_code)
    if path.exists():
        shutil.rmtree(path)

async def photo_add_main_collect(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.callback_query:
        query = update.callback_query
        await query.answer()
        data = query.data
        if data == "send_main_photos":
            buffer = context.user_data.get("main_photos_buffer", [])
            if not buffer:
                await query.edit_message_text("❌ Нет фото для сохранения. Отправьте хотя бы одно фото.")
                return PHOTO_ADD_MAIN_COLLECT
            prod_code = context.user_data["photo_prod_code"]
            saved = 0
            for file_obj, _ in buffer:
                if await save_photo_from_telegram(file_obj, prod_code, "main"):
                    saved += 1
            context.user_data.pop("main_photos_buffer", None)
            await query.edit_message_text(f"✅ Сохранено основных фото: {saved}.")
            await show_product_edit(update, context, context.user_data["photo_prod_id"])
            return ConversationHandler.END
        elif data == "cancel_main_photos":
            context.user_data.pop("main_photos_buffer", None)
            await query.edit_message_text("❌ Добавление фото отменено.")
            await show_photo_management(update, context, context.user_data["photo_prod_id"])
            return PHOTO_MANAGE
        else:
            await query.edit_message_text("Неизвестная команда.")
            return PHOTO_ADD_MAIN_COLLECT
    if not update.message.photo:
        await update.message.reply_text("❌ Пожалуйста, отправьте фото. Или нажмите кнопку «Отправить фото» для завершения.",
                                        reply_markup=InlineKeyboardMarkup([
                                            [InlineKeyboardButton("✅ Отправить фото", callback_data="send_main_photos")],
                                            [InlineKeyboardButton("❌ Отменить", callback_data="cancel_main_photos")]
                                        ]))
        return PHOTO_ADD_MAIN_COLLECT
    try:
        photo_file = await download_photo_with_retry(update.message.photo[-1])
        context.user_data.setdefault("main_photos_buffer", []).append((photo_file, update.message))
        current_count = len(context.user_data["main_photos_buffer"])
        msg_id = context.user_data.get("photo_main_message_id")
        if msg_id:
            try:
                await context.bot.edit_message_text(
                    chat_id=update.effective_chat.id, message_id=msg_id,
                    text=f"📸 *Добавление основных фото*\n\n📷 Фото добавлено. Всего в буфере: {current_count}.\n\n"
                         "Отправьте ещё фото или нажмите кнопку «Отправить фото» для сохранения.",
                    parse_mode="Markdown",
                    reply_markup=InlineKeyboardMarkup([
                        [InlineKeyboardButton("✅ Отправить фото", callback_data="send_main_photos")],
                        [InlineKeyboardButton("❌ Отменить", callback_data="cancel_main_photos")]
                    ])
                )
            except Exception as e:
                logger.warning(f"Не удалось обновить сообщение: {e}")
        else:
            await update.message.reply_text(f"📷 Фото добавлено. Всего в буфере: {current_count}.\n"
                                            "Нажмите кнопку «Отправить фото» для сохранения.",
                                            reply_markup=InlineKeyboardMarkup([
                                                [InlineKeyboardButton("✅ Отправить фото", callback_data="send_main_photos")],
                                                [InlineKeyboardButton("❌ Отменить", callback_data="cancel_main_photos")]
                                            ]))
    except Exception as e:
        logger.error(f"Ошибка загрузки фото: {e}")
        await update.message.reply_text("❌ Не удалось скачать фото. Попробуйте ещё раз.")

# ------------------------- Отображение товаров с пагинацией -------------------------
async def show_products(update: Update, context: ContextTypes.DEFAULT_TYPE,
                        search: str = None, type_id: int = None, flag_ids: List[int] = None,
                        page: int = 0):
    PER_PAGE = 8
    offset = page * PER_PAGE
    products, total = await db.get_products(search=search, type_id=type_id, flag_ids=flag_ids,
                                            offset=offset, limit=PER_PAGE)
    total_pages = (total + PER_PAGE - 1) // PER_PAGE if total > 0 else 1
    if page >= total_pages:
        page = total_pages - 1
        offset = page * PER_PAGE
        products, total = await db.get_products(search=search, type_id=type_id, flag_ids=flag_ids,
                                                offset=offset, limit=PER_PAGE)
    text = "*Список товаров*"
    if search:
        text += f" (поиск: `{search}`)"
    if type_id:
        t = await db.get_product_type_by_id(type_id)
        text += f" (тип: {t['name'] if t else type_id})"
    if flag_ids:
        text += f" (флаги: {len(flag_ids)})"
    text += f"\nСтраница {page+1} из {max(1, total_pages)}\n\n"
    if not products:
        text += "Нет товаров, удовлетворяющих условиям."
    else:
        for p in products:
            text += f"`{p['id']}` – {p['name']} (код: {p['code']})\n"
    keyboard = []
    for p in products:
        keyboard.append([
            InlineKeyboardButton(f"📄 {p['name']} (ID {p['id']})", callback_data=f"prod_details_{p['id']}"),
            InlineKeyboardButton("✏️", callback_data=f"prod_edit_{p['id']}")
        ])
    nav = []
    if page > 0:
        nav.append(InlineKeyboardButton("◀ Назад", callback_data=f"products_page_{page-1}"))
    if page < total_pages - 1:
        nav.append(InlineKeyboardButton("Вперёд ▶", callback_data=f"products_page_{page+1}"))
    if nav:
        keyboard.append(nav)
    keyboard.append([InlineKeyboardButton("➕ Добавить товар", callback_data="prod_add")])
    keyboard.append([InlineKeyboardButton("🔍 Новый поиск", callback_data="menu_search")])
    keyboard.append([InlineKeyboardButton("🏠 Главное меню", callback_data="main_menu")])
    context.user_data["products_search"] = search
    context.user_data["products_type_id"] = type_id
    context.user_data["products_flag_ids"] = flag_ids
    await send_or_edit_message(update, context, text, InlineKeyboardMarkup(keyboard), new_message=False)

def register_handlers(app: Application):

    app.add_handler(CallbackQueryHandler(main_menu_callback, pattern="^main_menu$"))
    app.add_handler(CallbackQueryHandler(menu_products_callback, pattern="^menu_products$"))
    app.add_handler(CallbackQueryHandler(menu_flags_callback, pattern="^menu_flags$"))
    app.add_handler(CallbackQueryHandler(menu_prod_types_callback, pattern="^menu_prod_types$"))
    app.add_handler(CallbackQueryHandler(menu_mat_types_callback, pattern="^menu_mat_types$"))

    app.add_handler(CallbackQueryHandler(menu_search_callback, pattern="^menu_search$"))

    # Пагинация товаров
    app.add_handler(CallbackQueryHandler(products_page_callback, pattern="^products_page_\\d+$"))
    # Просмотр и редактирование товаров
    app.add_handler(CallbackQueryHandler(product_details_callback, pattern="^prod_details_\\d+$"))

        per_message=False,

    app.add_handler(add_prod_conv)

    # Редактирование названия, кода, типа товара
    edit_name_conv = ConversationHandler(
        entry_points=[CallbackQueryHandler(edit_product_name_entry, pattern="^prod_edit_name_\\d+$")],

        states={EDIT_PROD_TYPE_ID: [MessageHandler(filters.TEXT & ~filters.COMMAND, edit_product_type_id)]},
        fallbacks=[CommandHandler("cancel", cancel), CallbackQueryHandler(cancel, pattern="^main_menu$")],
        per_message=False,
    )
    app.add_handler(edit_type_conv)

    # Редактирование флагов
    edit_flags_conv = ConversationHandler(
        entry_points=[CallbackQueryHandler(edit_product_flags_start, pattern="^prod_edit_flags_\\d+$")],
        states={EDIT_PROD_FLAGS: [
            CallbackQueryHandler(edit_product_toggle_flag, pattern="^prod_toggle_flag_\\d+_\\d+$"),

        fallbacks=[CommandHandler("cancel", cancel), CallbackQueryHandler(cancel, pattern="^main_menu$")],
        per_message=False,

CREATE TABLE products_type (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- ==================== products ====================
CREATE TABLE products (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    code VARCHAR(100) NOT NULL UNIQUE,
    name VARCHAR(200) NOT NULL,
    products_type_id INTEGER NOT NULL,
    CONSTRAINT fk_products_type
        FOREIGN KEY (products_type_id)
        REFERENCES products_type(id)
        ON DELETE RESTRICT
);

-- ==================== flags ====================
CREATE TABLE flags (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

-- ==================== flags_for_products ====================
CREATE TABLE flags_for_products (
    products_id INTEGER NOT NULL,
    flags_id INTEGER NOT NULL,
    PRIMARY KEY (products_id, flags_id),
    CONSTRAINT fk_ffp_product
        FOREIGN KEY (products_id)
        REFERENCES products(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_ffp_flag
        FOREIGN KEY (flags_id)
        REFERENCES flags(id)
        ON DELETE CASCADE
);

-- ==================== materials_type ====================
CREATE TABLE materials_type (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- ==================== materials ====================
CREATE TABLE materials (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    code VARCHAR(100) NOT NULL UNIQUE,
    name VARCHAR(200) NOT NULL,
    materials_type_id INTEGER NOT NULL,
    CONSTRAINT fk_materials_type
        FOREIGN KEY (materials_type_id)
        REFERENCES materials_type(id)
        ON DELETE RESTRICT
);

-- ==================== materials_for_products ====================
CREATE TABLE materials_for_products (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    products_id INTEGER NOT NULL,
    materials_type_id INTEGER NOT NULL,
    cost INTEGER NOT NULL CHECK (cost >= 0),
    CONSTRAINT fk_mfp_product
        FOREIGN KEY (products_id)
        REFERENCES products(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_mfp_material_type
        FOREIGN KEY (materials_type_id)
        REFERENCES materials_type(id)
        ON DELETE RESTRICT,
    CONSTRAINT uq_product_material_type UNIQUE(products_id, materials_type_id)
);

Назначение

Самая важная бизнес-таблица.

Что хранит
какие материалы доступны;
стоимость изготовления;
настройки производства.
Структура
Поле	Тип
id	INTEGER
products_id	INTEGER
materials_type_id	INTEGER
cost	INTEGER
Логика

Пример:

Корпус Arduino:
PLA → 500₽
PETG → 900₽
ABS → 1200₽
Почему materials_type, а не materials

Потому что:

стоимость обычно зависит от типа материала

а не цвета.

cost
INTEGER
Почему INTEGER

Для рублей:

500
1200
9999
CHECK(cost >= 0)

Запрещает:

-500
UNIQUE(products_id, materials_type_id)

Запрещает:

товар + PLA
товар + PLA

products	flags	MANY TO MANY

===== Файл: telegram_admin_panel\описание бота.txt =====
Общая архитектура Telegram‑бота для интернет‑магазина
Бот написан на Python с использованием библиотеки python-telegram-bot (версия 20+). Для работы с PostgreSQL используется пул соединений asyncpg, асинхронность реализована через asyncio. Все фотографии товаров обрабатываются через Pillow (PIL) и конвертируются в формат .webp для оптимизации.

Структура проекта и файлы
script.py – основной исполняемый файл бота.

token.txt – содержит токен Telegram‑бота.

config_db.txt – параметры подключения к PostgreSQL (host, port, database, user, password).

admin_ids.txt – список числовых ID администраторов (по одному на строку). Если файл отсутствует, бот становится доступным любому пользователю (небезопасно).

Папка media/products/<код_товара>/ – хранит фото каждого товара:

preview.webp – уменьшенная и сжатая (качество 50%) превьюшка.

size.webp – схема или размерный ряд (качество 85%).

1.webp, 2.webp, … – основные фотографии. Нумерация автоматически поддерживается и перенумеровывается при удалении.

Схема базы данных (по коду)
Таблицы (имена соответствуют реальным):

products – id, code (уникальный), name, products_type_id.

products_type – id, name.

flags – id, name.

flags_for_products – связь многие‑ко‑многим: products_id, flags_id.

materials_type – id, name (например, «дерево», «металл»).

materials_for_products – id, products_id, materials_type_id, cost (цена для данного материала).

Все операции с БД инкапсулированы в классе Database – он получает пул соединений и предоставляет асинхронные методы для выборки, вставки, обновления и каскадного удаления.

Ключевые компоненты архитектуры
Декоратор @admin_only – проверяет, что ID пользователя из update.effective_user.id присутствует в множестве ADMIN_IDS. Если нет – отвечает отказом.

ConversationHandler – для многошаговых сценариев (добавление товара, редактирование полей, управление фото, расширенный поиск).

Работа с фото – функции:

download_photo_with_retry – 3 попытки скачать файл с паузой 2 секунды.

save_photo_from_telegram – конвертация и сохранение в WEBP.

delete_photo, delete_media_folder, move_media_folder – управление файлами.

send_product_photos – отправка медиагруппой всех фото товара.

Пагинация – везде, где список может быть длинным (товары, флаги, типы товаров, типы материалов), используется постраничный вывод по 8 элементов с кнопками «Вперёд / Назад».

Увеличенные таймауты – через HTTPXRequest заданы connect_timeout=30, read_timeout=120 для надёжной загрузки больших фото.

Полный функционал админ‑панели
1. Главное меню
После команды /start или нажатия «Главное меню» появляется клавиатура:

📦 Товары – управление товарами.

🏷️ Флаги / Теги – справочник флагов (характеристик товара).

📂 Типы товаров – справочник категорий.

🧱 Типы материалов – справочник материалов для расчёта стоимостей.

🔍 Поиск товаров – два режима поиска.

Все действия доступны только администраторам, чьи ID перечислены в admin_ids.txt.

2. Управление товарами
2.1 Просмотр списка товаров
Отображается таблица с ID, названием, кодом.

Каждая строка имеет две кнопки:

📄 «Название (ID)» – подробный просмотр без возможности редактирования.

✏️ «Редактировать» – переход в полноценный редактор.

Внизу – пагинация (8 товаров на страницу), кнопка «➕ Добавить товар», «🔍 Новый поиск» и возврат в главное меню.

2.2 Добавление товара (пошаговый диалог)
Ввод названия товара.

Ввод уникального кода – проверка на дубликат.

Выбор типа товара – отправкой ID из списка (справочник типов товаров).

После создания товара автоматически создаётся пустая папка media/products/<код>/, и бот переходит в режим редактирования нового товара.

2.3 Редактирование товара (все поля)
В редакторе отображаются текущие значения и доступны кнопки:

✏️ Название – новый текст.

✏️ Код – изменение кода приводит к автоматическому переименованию папки с фотографиями (move_media_folder).

✏️ Тип товара – смена категории.

🏷️ Флаги – переключатели для каждого флага (✅/❌), сохранение.

💰 Управление стоимостью – подменю для добавления/изменения/удаления цен по типам материалов.

📸 Управление фото – подменю для добавления, удаления и просмотра фото.

🗑️ Удалить товар – каскадное удаление (флаги, стоимости, папка с фото).

2.4 Работа с фото товара
Подменю «Управление фото»:

➕ Добавить фото – выбор типа: Preview, Size, Основное фото.

Для Preview / Size – отправляется одно фото, оно сохраняется с фиксированным именем (preview.webp / size.webp).

Для основных фото – можно отправить несколько фото (альбомом или по одному). Они добавляются в буфер, а после нажатия кнопки «Отправить фото» сохраняются с автоматической нумерацией (1.webp, 2.webp…).

🗑️ Удалить фото – список всех существующих фото (Preview, Size, каждое основное), выбираете – удаляется, основные фото перенумеровываются заново.

🖼️ Посмотреть фото – отправляет все фото товара одной медиагруппой (максимум 10 фото за раз).

2.5 Управление стоимостью по материалам
➕ Добавить / изменить стоимость – выбор типа материала из справочника, затем ввод целого числа (рубли). При повторном выборе того же материала цена перезаписывается.

🗑️ Удалить стоимость – выбор конкретной записи (material_name – цена) для удаления.

2.6 Режим «Подробнее» (без редактирования)
При нажатии на кнопку с названием товара в списке открывается карточка с полной информацией:

Название, код, тип, все привязанные флаги, все стоимости по материалам.

Кнопка «🖼️ Посмотреть фото» – отправляет все фото товара.

Кнопки возврата к списку или в главное меню.

3. Управление справочниками
Все три справочника (флаги, типы товаров, типы материалов) имеют унифицированный интерфейс:

Постраничный список (8 записей на страницу).

Каждая запись: кнопка «✏️ Название» (редактирование) и кнопка «🗑️» (удаление с подтверждением).

Внизу: пагинация, кнопка «➕ Добавить», возврат в главное меню.

3.1 Флаги
Добавление/редактирование – просто ввод названия.

Удаление – только удаление самого флага (он автоматически снимается со всех товаров, связь в flags_for_products стирается).

3.2 Типы товаров
Добавление/редактирование названия.

Удаление типа товара – каскадное: сначала удаляются все товары этого типа (вместе с их флагами, стоимостями, папками фото), затем сам тип. Выдаётся строгое предупреждение.

3.3 Типы материалов
Добавление/редактирование названия.

Удаление типа материала – удаляются все записи materials_for_products, связанные с этим типом, а затем сам тип. Стоимости товаров по этому материалу исчезают.

4. Поиск товаров
4.1 Поиск по ключевому слову
Администратор вводит слово (или часть слова).

Бот ищет среди name и code (регистронезависимо, через ILIKE).

Результат выводится в том же формате, что и обычный список товаров (с пагинацией).

4.2 Расширенный поиск (по типу и флагам)
Это пошаговый диалог:

Шаг 1 – выбор типа товара
Выводится список типов товаров с пагинацией. Каждый тип можно выбрать/снять (✓/✗). Выбрать можно только один тип. Кнопка «Пропустить» – поиск по всем типам. Кнопка «Далее → выбор флагов».

Шаг 2 – выбор флагов
Выводится список всех флагов с пагинацией. Флаги выбираются множественно (переключатели). Кнопка «Сбросить все флаги» – очистить выбор. Кнопка «Поиск» – выполнить запрос.

Логика поиска:

Если выбраны флаги – ищутся товары, у которых присутствуют ВСЕ выбранные флаги.

Если тип выбран – фильтр по products_type_id.

Если ничего не выбрано – показываются все товары (аналогично меню «Товары»).

После выполнения поиска открывается стандартный список товаров с пагинацией, при этом параметры поиска сохраняются в context.user_data для корректной работы пагинации.

Детали реализации, важные для администрирования
Безопасность и права доступа
Файл admin_ids.txt – если его нет, бот разрешает доступ всем (логируется предупреждение). Рекомендуется всегда создавать этот файл хотя бы с одним ID.

Декоратор @admin_only оборачивает все callback-обработчики и команду /start. Любое действие вне ConversationHandler проверяет права.

Транзакции и каскадное удаление
При удалении типа товара или товара все изменения выполняются в одной транзакции (async with conn.transaction()). Это гарантирует, что если что-то пойдёт не так, база не останется в противоречивом состоянии.

Удаление папок с фото происходит после удаления записей из БД, но перед фиксацией транзакции – чтобы в случае ошибки файлы не потерялись.

Работа с фото – повторные попытки и таймауты
download_photo_with_retry – 3 попытки, пауза 2 секунды. Особенно полезно при нестабильном интернете или больших файлах.

Таймауты HTTPXRequest увеличены до 30 с на соединение и 120 с на чтение. Это предотвращает ошибку TimedOut при загрузке тяжёлых фото от пользователя.

Пагинация и сохранение контекста
Для каждого справочника в context.user_data хранятся текущая страница, ID редактируемого элемента и т.п.

При пагинации списка товаров сохраняются параметры поиска (products_search, products_type_id, products_flag_ids), чтобы после нажатия «Вперёд/Назад» фильтрация не сбрасывалась.

Удобство администратора
Все длительные операции (например, сохранение альбома фото) имеют промежуточные уведомления и кнопку отмены.

Редактирование флагов на лету (без выхода из режима) – вы можете понажимать на кнопки переключения, а потом сохранить одним нажатием.

При удалении основного фото все оставшиеся основные фото автоматически перенумеровываются заново (1, 2, 3…), чтобы не было разрывов.

Вывод
Бот представляет собой полноценную админ‑панель для интернет‑магазина в Telegram. Он позволяет:

Управлять товарами, их атрибутами (флаги, типы, материалы/цены) и мультимедиа.

Гибко искать товары как по ключевому слову, так и по комбинации типа и обязательных флагов.

Поддерживать справочники (типы товаров, флаги, материалы) с единообразным интерфейсом редактирования.

Безопасно удалять сущности с каскадным влиянием на связанные данные и файлы.

Архитектура спроектирована с расчётом на асинхронность, устойчивость к сбоям при загрузке фото и удобство для администратора, который управляет ассортиментом прямо из мессенджера.

<script>
    const gridContainer = document.getElementById('catalog-grid');
    const loader = document.getElementById('catalog-loader');
    const errorBlock = document.getElementById('catalog-error');
    const typeSelect = document.getElementById('type-select');
    const flagsContainer = document.getElementById('flags-checkboxes');

    let currentTimeout = null;

    function escapeHtml(str) {
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        });
    }

    function renderProducts(products) {
        if (!products.length) {
            gridContainer.innerHTML = '<p class="empty">Товаров не найдено</p>';
            gridContainer.style.display = 'grid';
            return;
        }
        let html = '';
        for (const prod of products) {
            const imagePath = `/media/products/${encodeURIComponent(prod.code)}/preview.webp`;
            html += `
                <div class="slider-items">
                    <img src="${imagePath}" alt="${escapeHtml(prod.name)}" loading="lazy" onerror="this.src='/media/interface/placeholder.png'">
                    <div class="product-name">${escapeHtml(prod.name)}</div>
                    <div class="code-name">Код: ${escapeHtml(prod.code)}</div>
                </div>
            `;
        }
        gridContainer.innerHTML = html;
        gridContainer.style.display = 'grid';
    }

    async function loadProducts() {
        const typeId = typeSelect.value;
        const selectedFlags = Array.from(document.querySelectorAll('.flag-checkbox:checked')).map(cb => cb.value);

        let url = '/api/products?';
        const params = new URLSearchParams();
        if (typeId) params.append('type_id', typeId);
        selectedFlags.forEach(fid => params.append('flag_ids', fid));
        url += params.toString();

        try {
            loader.style.display = 'block';
            gridContainer.style.display = 'none';
            errorBlock.style.display = 'none';
            const response = await fetch(url);
            if (!response.ok) throw new Error(`HTTP ${response.status}`);
            const products = await response.json();
            renderProducts(products);
        } catch (err) {
            console.error(err);
            loader.style.display = 'none';
            errorBlock.style.display = 'block';
        } finally {
            loader.style.display = 'none';
        }
    }

    async function loadFilters() {
        try {
            const typesResp = await fetch('/api/types');
            if (typesResp.ok) {
                const types = await typesResp.json();
                typeSelect.innerHTML = '<option value="">Все типы</option>';
                types.forEach(t => {
                    const option = document.createElement('option');
                    option.value = t.id;
                    option.textContent = t.name;
                    typeSelect.appendChild(option);
                });
            }

            const flagsResp = await fetch('/api/flags');
            if (flagsResp.ok) {
                const flags = await flagsResp.json();
                flagsContainer.innerHTML = '';
                flags.forEach(f => {
                    const label = document.createElement('label');
                    label.className = 'flag-checkbox-label';
                    const cb = document.createElement('input');
                    cb.type = 'checkbox';
                    cb.value = f.id;
                    cb.className = 'flag-checkbox';
                    label.appendChild(cb);
                    label.appendChild(document.createTextNode(` ${f.name}`));
                    flagsContainer.appendChild(label);
                });
            }

            await loadProducts();
        } catch (err) {
            console.error('Ошибка загрузки фильтров:', err);
        }
    }

    function onChangeFilter() {
        if (currentTimeout) clearTimeout(currentTimeout);
        currentTimeout = setTimeout(() => {
            loadProducts();
        }, 300);
    }

    typeSelect.addEventListener('change', onChangeFilter);
    flagsContainer.addEventListener('change', (e) => {
        if (e.target.classList.contains('flag-checkbox')) {
            onChangeFilter();
        }
    });

    // Модальное окно
    const modal = document.getElementById('aboutModal');
    const aboutLink = document.getElementById('aboutLink');
    const closeSpan = document.querySelector('.close');
    aboutLink.onclick = (e) => { e.preventDefault(); modal.style.display = 'block'; };
    closeSpan.onclick = () => { modal.style.display = 'none'; };
    window.onclick = (e) => { if (e.target === modal) modal.style.display = 'none'; };

    loadFilters();
</script>
