-- #########################################################################
-- # LUTO SHOP DATABASE
-- # PostgreSQL Production Structure (с материалами, заказами и user_id)
-- #########################################################################

DROP TABLE IF EXISTS order_items CASCADE;
DROP TABLE IF EXISTS orders CASCADE;
DROP TABLE IF EXISTS flags_for_products CASCADE;
DROP TABLE IF EXISTS materials_for_products CASCADE;
DROP TABLE IF EXISTS product_main_features CASCADE;
DROP TABLE IF EXISTS product_extra_features CASCADE;

DROP TABLE IF EXISTS flags CASCADE;
DROP TABLE IF EXISTS materials CASCADE;
DROP TABLE IF EXISTS materials_type CASCADE;
DROP TABLE IF EXISTS main_features_types CASCADE;

DROP TABLE IF EXISTS products CASCADE;
DROP TABLE IF EXISTS products_type CASCADE;
DROP TABLE IF EXISTS users CASCADE;

-- ==================== users ====================
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    phone VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(256) NOT NULL,
    full_name VARCHAR(200),
    email VARCHAR(200),
    address TEXT,
    contact_time VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ==================== products_type ====================
CREATE TABLE products_type (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- ==================== products ====================
CREATE TABLE products (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    code VARCHAR(100) NOT NULL UNIQUE,
    name VARCHAR(200) NOT NULL,
    products_type_id INTEGER NOT NULL,
    CONSTRAINT fk_products_type
        FOREIGN KEY (products_type_id)
        REFERENCES products_type(id)
        ON DELETE RESTRICT
);

-- ==================== flags ====================
CREATE TABLE flags (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

-- ==================== flags_for_products ====================
CREATE TABLE flags_for_products (
    products_id INTEGER NOT NULL,
    flags_id INTEGER NOT NULL,
    PRIMARY KEY (products_id, flags_id),
    CONSTRAINT fk_ffp_product
        FOREIGN KEY (products_id)
        REFERENCES products(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_ffp_flag
        FOREIGN KEY (flags_id)
        REFERENCES flags(id)
        ON DELETE CASCADE
);

-- ==================== materials_type ====================
CREATE TABLE materials_type (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- ==================== materials ====================
CREATE TABLE materials (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    code VARCHAR(100) NOT NULL UNIQUE,
    name VARCHAR(200) NOT NULL,
    materials_type_id INTEGER NOT NULL,
    CONSTRAINT fk_materials_type
        FOREIGN KEY (materials_type_id)
        REFERENCES materials_type(id)
        ON DELETE RESTRICT
);

-- ==================== materials_for_products ====================
CREATE TABLE materials_for_products (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    products_id INTEGER NOT NULL,
    materials_type_id INTEGER NOT NULL,
    cost INTEGER NOT NULL CHECK (cost >= 0),
    CONSTRAINT fk_mfp_product
        FOREIGN KEY (products_id)
        REFERENCES products(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_mfp_material_type
        FOREIGN KEY (materials_type_id)
        REFERENCES materials_type(id)
        ON DELETE RESTRICT,
    CONSTRAINT uq_product_material_type UNIQUE(products_id, materials_type_id)
);

-- ==================== Основные характеристики ====================
CREATE TABLE main_features_types (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE product_main_features (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    products_id INTEGER NOT NULL,
    feature_id INTEGER NOT NULL,
    value TEXT NOT NULL,
    CONSTRAINT fk_pmf_product
        FOREIGN KEY (products_id)
        REFERENCES products(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_pmf_feature
        FOREIGN KEY (feature_id)
        REFERENCES main_features_types(id)
        ON DELETE CASCADE,
    CONSTRAINT uq_product_feature UNIQUE(products_id, feature_id)
);

CREATE TABLE product_extra_features (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    products_id INTEGER NOT NULL,
    name VARCHAR(100) NOT NULL,
    value TEXT NOT NULL,
    CONSTRAINT fk_pef_product
        FOREIGN KEY (products_id)
        REFERENCES products(id)
        ON DELETE CASCADE
);

-- ==================== ЗАКАЗЫ ====================
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    session_id VARCHAR(100) NOT NULL,
    user_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
    user_name VARCHAR(200) NOT NULL,
    phone VARCHAR(50) NOT NULL,
    email VARCHAR(200),
    address TEXT,
    comment TEXT,
    contact_time VARCHAR(100),
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'Ожидает подтверждения'
);

CREATE TABLE order_items (
    id SERIAL PRIMARY KEY,
    order_id INTEGER REFERENCES orders(id) ON DELETE CASCADE,
    product_code VARCHAR(100) NOT NULL,
    product_name VARCHAR(200) NOT NULL,
    material_code VARCHAR(100) NOT NULL,
    material_name VARCHAR(200) NOT NULL,
    cost INTEGER NOT NULL,
    quantity INTEGER DEFAULT 1
);

-- ==================== ИНДЕКСЫ ====================
CREATE INDEX idx_products_code ON products(code);
CREATE INDEX idx_products_type ON products(products_type_id);
CREATE INDEX idx_materials_code ON materials(code);
CREATE INDEX idx_flags_name ON flags(name);
CREATE INDEX idx_materials_type_name ON materials_type(name);
CREATE INDEX idx_mfp_product ON materials_for_products(products_id);
CREATE INDEX idx_ffp_product ON flags_for_products(products_id);
CREATE INDEX idx_main_features_types_name ON main_features_types(name);
CREATE INDEX idx_pmf_product ON product_main_features(products_id);
CREATE INDEX idx_pef_product ON product_extra_features(products_id);
CREATE INDEX idx_orders_session ON orders(session_id);
CREATE INDEX idx_orders_date ON orders(order_date);
CREATE INDEX idx_orders_user ON orders(user_id);
